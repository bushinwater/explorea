================================================================
EXPLOREA — API
================================================================

----------------------------------------------------------------
AUTH
----------------------------------------------------------------

POST /api/auth/register
Body:     { firstName, lastName, email, password, confirmPassword }
Response: { accessToken, refreshToken, needsInterests, user: { id, firstName, lastName, email, role, travelerRank } }

POST /api/auth/login
Body:     { email, password }
Response: { accessToken, refreshToken, needsInterests, user: { id, firstName, lastName, email, role, travelerRank } }

POST /api/auth/refresh
Body:     { refreshToken }
Response: { accessToken, refreshToken }

POST /api/auth/logout
Body:     { refreshToken }
Response: 204

POST /api/auth/forgot-password
Body:     { email }
Response: 200

POST /api/auth/reset-password
Body:     { token, newPassword }
Response: 200

KORISNIK
----------------------------------------------------------------

GET /api/users/me
Response: {
  id, firstName, lastName, email,
  travelerRank, totalVisits, memberSince,
  preferences: { theme, language, notificationsEnabled }
}

PUT /api/users/me
Body:     { firstName, lastName }
Response: UserProfileDto

PUT /api/users/me/preferences
Body:     { theme, language, notificationsEnabled }
Response: 200

GET /api/users/me/interests
Response: [{ id, name }]

POST /api/users/me/interests
Body:     { interestTagIds: [1, 3, 5] }
Response: 200

HOME SCREEN
----------------------------------------------------------------

GET /api/home
Response: {
  heroBanners:          [{ id, title, imageUrl, linkType, linkSlug }],
  featuredDestinations: [DestinationCardDto],
  upcomingEvents:       [EventCardDto],
  popularActivities:    [ActivityCardDto],
  travelerReviews:      [ReviewDto]
}

----------------------------------------------------------------
DESTINACIJE
----------------------------------------------------------------

GET /api/destinations?cityId=&regionId=&featured=&page=1&pageSize=10
Response: PagedResult<DestinationCardDto>

DestinationCardDto: {
  id, name, slug, previewImageUrl,
  avgRating, city: { id, name },
  location: { lat, lng }
}

GET /api/destinations/{slug}
Response: {
  id, name, slug, description, avgRating,
  city: { id, name },
  location: { lat, lng },
  images:               [{ imageUrl, sortOrder }],
  places:               [PlaceCardDto],
  events:               [EventCardDto],
  activities:           [ActivityCardDto],
  reviews:              [ReviewDto],
  similarDestinations:  [DestinationCardDto]
}

----------------------------------------------------------------
OBJEKTI (Places)
----------------------------------------------------------------

GET /api/places?destinationId=&cityId=&typeId=&minRating=&isOpen=&page=1&pageSize=10
Response: PagedResult<PlaceCardDto>

PlaceCardDto: {
  id, name, slug, type: { id, name },
  avgRating, address, distanceKm,
  previewImageUrl, isOpen, priceRange
}

GET /api/places/{slug}
Response: {
  id, name, slug, description, address, phone, websiteUrl,
  avgRating, priceRange,
  type:        { id, name },
  location:    { lat, lng },
  images:      [{ imageUrl, caption, sortOrder }],
  amenities:   [{ id, name }],
  workingDays: [{ dayOfWeek, isClosed, intervals: [{ opensAt, closesAt }] }],
  events:      [EventCardDto],
  reviews:     [ReviewDto],
  similarPlaces: [PlaceCardDto]
}

----------------------------------------------------------------
DOGAĐAJI
----------------------------------------------------------------

GET /api/events?cityId=&typeId=&from=&to=&free=&page=1&pageSize=10
Response: PagedResult<EventCardDto>

EventCardDto: {
  id, name, slug, type: { id, name },
  startDatetime, endDatetime, price,
  address, city: { id, name },
  previewImageUrl, isFeatured
}

GET /api/events/{slug}
Response: {
  id, name, slug, description,
  type:          { id, name },
  startDatetime, endDatetime, price, ticketUrl,
  address,
  location:      { lat, lng },
  images:        [{ imageUrl, caption, sortOrder }],
  reviews:       [ReviewDto],
  similarEvents: [EventCardDto]
}

GET /api/events/calendar?year=2024&month=7
Response: [{ date, count, events: [EventCardDto] }]

----------------------------------------------------------------
AKTIVNOSTI
----------------------------------------------------------------

GET /api/activities?destinationId=&categoryId=&seasonId=&difficulty=&page=1&pageSize=10
Response: PagedResult<ActivityCardDto>

ActivityCardDto: {
  id, name, slug, category: { id, name },
  avgRating, durationMinutes, price,
  difficultyLevel, previewImageUrl
}

GET /api/activities/{slug}
Response: {
  id, name, slug, description,
  category:    { id, name },
  season:      { id, name },
  price, durationMinutes, difficultyLevel,
  distanceKm, startPoint, extraData,
  location:    { lat, lng },
  inclusions:  [{ id, item }],
  images:      [{ imageUrl, sortOrder }],
  reviews:     [ReviewDto],
  similarActivities: [ActivityCardDto]
}
